 STEP 3: FIREWALL SETUP (UFW)
═══════════════════════════════════════════════════════════════════════════════

  sudo ufw default deny incoming
  sudo ufw default allow outgoing
  sudo ufw allow 22/tcp      # SSH — NEVER BLOCK THIS
  sudo ufw allow 80/tcp      # HTTP
  sudo ufw allow 443/tcp     # HTTPS
  sudo ufw allow 8000/tcp    # FastAPI backend (adjust if your app uses different port)
  sudo ufw allow 3000/tcp    # React dev server (if needed)
  sudo ufw allow 8080/tcp    # ORDS / alternative port
  sudo ufw enable

  # Verify:
  sudo ufw status verbose

═══════════════════════════════════════════════════════════════════════════════
 STEP 4: ORACLE INSTANT CLIENT (For Database Connectivity)
═══════════════════════════════════════════════════════════════════════════════

  # Create directory
  sudo mkdir -p /opt/oracle
  cd /opt/oracle

  # Download Oracle Instant Client 23.4 (matches 26ai)
  sudo wget https://download.oracle.com/otn_software/linux/instantclient/2340000/instantclient-basic-linux.x64-23.4.0.24.05.zip

  sudo unzip instantclient-basic-linux.x64-23.4.0.24.05.zip

  # Install required library
  sudo apt install -y libaio1

  # Set environment variables (PERMANENT)
  echo 'export ORACLE_HOME=/opt/oracle/instantclient_23_4' >> ~/.bashrc
  echo 'export PATH=$ORACLE_HOME:$PATH' >> ~/.bashrc
  echo 'export LD_LIBRARY_PATH=$ORACLE_HOME:$LD_LIBRARY_PATH' >> ~/.bashrc
  echo 'export TNS_ADMIN=/home/ubuntu/oracle_wallet' >> ~/.bashrc

  # Apply immediately
  source ~/.bashrc

  # Method A: Using SQLcl (Oracle SQL Command Line) — RECOMMENDED

═══════════════════════════════════════════════════════════════════════════════
 STEP 10: ENABLE SCHEDULER JOBS (Autonomous Cognition)
═══════════════════════════════════════════════════════════════════════════════

  # Connect to database
  sqlcl ADMIN/Your_DB_Password@yourdb_high

  # Enable all scheduler jobs:
  BEGIN
    DBMS_SCHEDULER.enable('SNS_COGNITION_MASTER');
    DBMS_SCHEDULER.enable('SNS_INTROSPECTION_STREAM');
    DBMS_SCHEDULER.enable('SNS_MEMORY_CONSOLIDATION');
    DBMS_SCHEDULER.enable('SNS_EMOTIONAL_DECAY');
    DBMS_SCHEDULER.enable('SNS_GOAL_GENERATION');
  END;
  /

  # Verify jobs are running:
  SELECT job_name, state, last_start_date, next_run_date
  FROM user_scheduler_jobs
  WHERE job_name LIKE 'SNS_%';

  # Exit SQLcl
  exit

 STEP 15: SYSTEMD SERVICE (Auto-Start on Boot)
═══════════════════════════════════════════════════════════════════════════════

  # Create service file
  sudo nano /etc/systemd/system/7-agent.service

  # Paste this:
  ─────────────────────────────────────────
  [Unit]
  Description=7-Agent-1 Oracle Cognitive System
  After=network.target

  [Service]
  Type=simple
  User=ubuntu
  WorkingDirectory=/home/ubuntu/Oracle
  Environment=ORACLE_USER=ADMIN
  Environment=ORACLE_PASSWORD=Your_DB_Password_123
  Environment=ORACLE_DSN=yourdb_high
  Environment=PATH=/home/ubuntu/Oracle/venv/bin:/usr/local/bin:/usr/bin:/bin
  Environment=PYTHONPATH=/home/ubuntu/Oracle
  ExecStart=/home/ubuntu/Oracle/venv/bin/python3 /home/ubuntu/Oracle/main.py
  Restart=always
  RestartSec=5
  StandardOutput=journal
  StandardError=journal

  [Install]
  WantedBy=multi-user.target
  ─────────────────────────────────────────

  # Enable and start
  sudo systemctl daemon-reload
  sudo systemctl enable 7-agent
  sudo systemctl start 7-agent

  # Check status
  sudo systemctl status 7-agent

  # View logs
  sudo journalctl -u 7-agent -f

═══════════════════════════════════════════════════════════════════════════════
 STEP 16: NGINX REVERSE PROXY (Production)
═══════════════════════════════════════════════════════════════════════════════

  # Remove default site
  sudo rm /etc/nginx/sites-enabled/default

  # Create your config
  sudo nano /etc/nginx/sites-available/7-agent

  # Paste this:
  ─────────────────────────────────────────
  server {
      listen 80;
      server_name YOUR_DOMAIN_OR_IP;

      # Frontend (React build)
      location / {
          root /home/ubuntu/Oracle/front/dist;
          index index.html;
          try_files $uri $uri/ /index.html;
      }

      # Backend API (FastAPI)
      location /api/ {
          proxy_pass http://localhost:8000/;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection 'upgrade';
          proxy_set_header Host $host;
          proxy_set_header X-Real-IP $remote_addr;
          proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
          proxy_set_header X-Forwarded-Proto $scheme;
          proxy_cache_bypass $http_upgrade;
      }

      # WebSocket support (if needed)
      location /ws/ {
          proxy_pass http://localhost:8000/ws/;
          proxy_http_version 1.1;
          proxy_set_header Upgrade $http_upgrade;
          proxy_set_header Connection "upgrade";
      }
  }
  ─────────────────────────────────────────

  sudo ln -s /etc/nginx/sites-available/7-agent /etc/nginx/sites-enabled/
  sudo nginx -t
  sudo systemctl restart nginx

═══════════════════════════════════════════════════════════════════════════════
 STEP 18: QUICK TROUBLESHOOTING
═══════════════════════════════════════════════════════════════════════════════

  ❌ "ORA-12154: TNS:could not resolve the connect identifier"
  → Check tnsnames.ora in ~/oracle_wallet
  → Verify ORACLE_DSN matches the name in tnsnames.ora
  → Try full connection string instead of TNS name

  ❌ "ORA-28000: the account is locked"
  → Connect as ADMIN and unlock:
    ALTER USER your_user ACCOUNT UNLOCK;

  ❌ "DPI-1047: Cannot locate a 64-bit Oracle Client library"
  → Verify LD_LIBRARY_PATH includes /opt/oracle/instantclient_23_4
  → Run: sudo ldconfig (after adding to /etc/ld.so.conf.d/oracle.conf)

  ❌ "Module not found: bridge.engine_adapter"
  → Check that bridge/ directory exists with __init__.py
  → Verify PYTHONPATH includes /home/ubuntu/Oracle

  ❌ Scheduler jobs not running
  → Check: SELECT * FROM user_scheduler_job_run_details WHERE job_name LIKE 'SNS_%';
  → Verify job state: SELECT job_name, state FROM user_scheduler_jobs;
